"""Nine bounded same-evidence controls; no new source fetches or tool executions."""
import asyncio
import hashlib
import json
import sys
from dataclasses import asdict, replace
from pathlib import Path
from uuid import UUID, uuid4

root = Path('/tmp/cw2-q02-wire-eval')
sys.path[:0] = [str(root/'services/runtime/src'), str(root/'packages/protocol-python/src'), str(root/'packages/model-worker-sdk-python/src'), '/tmp/cw2-q02-eval-deps', str(root)]
from chatwaifu_runtime.agent.input_budget import fit_input_budget
from chatwaifu_runtime.agent.tool_calling import _FINAL_TOOL_POLICY
from chatwaifu_runtime.providers.contracts import LlmRequest, LlmInputBudget, LlmInputBudgetReport, LlmToolDefinition, LlmToolExchange, LlmToolCall, LlmToolResult
from chatwaifu_runtime.providers.openai_compatible import OpenAiCompatibleLlmProvider, build_messages, build_chat_completions_payload
from tools.runtime_source_evaluation import _RecordedProvider

def digest(value):
    return hashlib.sha256(json.dumps(value,ensure_ascii=False,sort_keys=True,default=str).encode()).hexdigest()

def restore(data):
    raw=dict(data)
    raw['generation_id']=UUID(raw['generation_id'])
    for field in ('context','history'):
        raw[field]=tuple(map(tuple,raw[field]))
    assert raw['images']==[]
    raw['images']=()
    raw['recalled_memory_texts']=tuple(raw['recalled_memory_texts'])
    raw['tools']=tuple(LlmToolDefinition(**t) for t in raw['tools'])
    raw['tool_exchanges']=tuple(LlmToolExchange(e['assistant_text'],tuple(LlmToolCall(**c) for c in e['calls']),tuple(LlmToolResult(**r) for r in e['results'])) for e in raw['tool_exchanges'])
    raw['input_budget']=LlmInputBudget(**raw['input_budget']) if raw['input_budget'] else None
    raw['input_budget_report']=LlmInputBudgetReport(**raw['input_budget_report']) if raw['input_budget_report'] else None
    return LlmRequest(**raw)

async def main():
    base=Path('/tmp/cw2-q02-wire-probe-20261003')
    rows=[json.loads(l) for l in (base/'actual-provider-payloads.jsonl').read_text().splitlines()]
    candidates=[r for r in rows if '移动电源' in r['request']['user_text'] and r['request']['tool_exchanges']]
    original=restore(candidates[-1]['request'])
    assert len(original.tool_exchanges)==3
    out=base/'answer-phase-controls';out.mkdir()
    key=json.loads(Path('/home/mubai/.local/share/chatwaifu-server/config/model-secrets.json').read_text())['chat']
    provider=_RecordedProvider(OpenAiCompatibleLlmProvider(base_url='https://mubai.website:8318/v1',model='gemini-3.8-flash-high',api_key=key,timeout_seconds=120,request_usage=True),out/'provider-rounds.jsonl',9)
    frozen_hash=digest([asdict(e) for e in original.tool_exchanges])
    metadata={'scope':'Same actual final source request; identical system/history/source bodies unless listed; tools never execute in this control','arms':{'native_auto':'original request','no_schema':'remove tools/schema and tool_choice only','closed_phase':'no_schema plus existing Runtime final-phase instruction'},'repeats':3,'maximum_provider_rounds':9,'source_exchanges_sha256':frozen_hash,'source_capture_sha256':hashlib.sha256((base/'actual-provider-payloads.jsonl').read_bytes()).hexdigest(),'model':'gemini-3.8-flash-high','source_tools_executed':0,'release_approved':False}
    (out/'metadata.json').write_text(json.dumps(metadata,ensure_ascii=False,indent=2)+'\n')
    for repeat in range(3):
        arms=['native_auto','no_schema','closed_phase']
        order=arms[repeat:]+arms[:repeat]
        for arm in order:
            request=replace(original,generation_id=uuid4(),input_budget_report=None)
            if arm!='native_auto':request=replace(request,tools=(),tool_choice='auto')
            if arm=='closed_phase':request=replace(request,system_prompt=request.system_prompt+_FINAL_TOOL_POLICY)
            request=fit_input_budget(request)
            assert digest([asdict(e) for e in request.tool_exchanges])==frozen_hash
            payload=build_chat_completions_payload('gemini-3.8-flash-high',request,build_messages(request),request_usage=True)
            sample=f'{arm}:r{repeat}'
            (out/f'{sample}-request.json').write_text(json.dumps(payload,ensure_ascii=False,indent=2)+'\n')
            provider.sample_key=sample
            error=None
            try:
                async for _event in provider.stream(request):pass
            except Exception as exc:error=type(exc).__name__
            call=provider.calls[-1]
            result={'sample_key':sample,'error_type':error,'raw_reply':call['text'],'finish_reason':call['finish_reason'],'requested_calls':call['requested_calls'],'usage':call['usage'],'latency_ms':call['latency_ms'],'input_budget_report':asdict(request.input_budget_report),'payload_sha256':digest(payload)}
            with (out/'results.jsonl').open('a') as f:f.write(json.dumps(result,ensure_ascii=False)+'\n')
            print(json.dumps({'sample':sample,'finish':call['finish_reason'],'characters':len(call['text']),'error':error}),flush=True)
asyncio.run(main())
