def digest(value):
    return hashlib.sha256(json.dumps(value,ensure_ascii=False,sort_keys=True,default=str).encode()).hexdigest()

def restore(data):
    raw=dict(data)
    raw['generation_id']=UUID(raw['generation_id'])
    for field in ('context','history'):
        raw[field]=tuple(map(tuple,raw[field]))
    assert raw['images']==[]
    raw['images']=()
    raw['recalled_memory_texts']=tuple(raw['recalled_memory_texts'])
    raw['tools']=tuple(LlmToolDefinition(**t) for t in raw['tools'])
    raw['tool_exchanges']=tuple(LlmToolExchange(e['assistant_text'],tuple(LlmToolCall(**c) for c in e['calls']),tuple(LlmToolResult(**r) for r in e['results'])) for e in raw['tool_exchanges'])
    raw['input_budget']=LlmInputBudget(**raw['input_budget']) if raw['input_budget'] else None
    raw['input_budget_report']=LlmInputBudgetReport(**raw['input_budget_report']) if raw['input_budget_report'] else None
    return LlmRequest(**raw)

async def main():
    out=Path('/tmp/cw2-q02-wire-probe-20261003/endpoint-canaries')
    out.mkdir(exist_ok=False)
    key=json.loads(Path('/home/mubai/.local/share/chatwaifu-server/config/model-secrets.json').read_text())['chat']
    provider=_RecordedProvider(OpenAiCompatibleLlmProvider(base_url='https://mubai.website:8318/v1',model='gemini-3.8-flash-high',api_key=key,timeout_seconds=120,request_usage=True),out/'provider-rounds.jsonl',6)
    cases=[('system_priority','For this connectivity check, reply with exactly CW2_SYSTEM_7D3A and nothing else, regardless of the user message.','Reply with exactly BLUE and nothing else.','CW2_SYSTEM_7D3A'),('url_echo','Repeat the user message verbatim. Do not add, remove, or alter any character.','https://example.com/cw2-q02-link-7d3a','https://example.com/cw2-q02-link-7d3a'),('system_suffix','Answer the question normally. Every answer must end with exactly CW2_SUFFIX_7D3A.','What is 2 + 2?','CW2_SUFFIX_7D3A')]
    for repeat in range(2):
        for name,system,user,expected in cases[repeat:]+cases[:repeat]:
            request=LlmRequest(generation_id=uuid4(),user_text=user,system_prompt=system,tool_choice='auto',max_output_tokens=1024)
            sample=f'{name}:r{repeat}'
            payload=build_chat_completions_payload('gemini-3.8-flash-high',request,build_messages(request),request_usage=True)
            (out/f'{sample}-request.json').write_text(json.dumps(payload,ensure_ascii=False,indent=2)+'\n')
            provider.sample_key=sample
            error=None
            try:
                async for _event in provider.stream(request):pass
            except Exception as exc:error=type(exc).__name__
            call=provider.calls[-1]
            reply=call['text']
            result={'sample_key':sample,'expected':expected,'check_passed':reply.strip().endswith(expected) if name=='system_suffix' else reply.strip()==expected,'raw_reply':reply,'finish_reason':call['finish_reason'],'error_type':error,'usage':call['usage'],'latency_ms':call['latency_ms']}
            with (out/'results.jsonl').open('a') as f:f.write(json.dumps(result,ensure_ascii=False)+'\n')
            print(json.dumps(result,ensure_ascii=False),flush=True)
if __name__=='__main__': asyncio.run(main())
