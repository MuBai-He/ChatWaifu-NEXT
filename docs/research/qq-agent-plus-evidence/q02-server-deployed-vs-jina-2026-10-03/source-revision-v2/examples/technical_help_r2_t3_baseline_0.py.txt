import asyncio

async def fetch_data(task_id: int):
    await asyncio.sleep(0.5)
    if task_id == 2:
        raise ValueError(f"任务 {task_id} 处理失败")
    return f"任务 {task_id} 成功返回的数据"

async def main():
    tasks = [fetch_data(i) for i in range(1, 4)]
    
    # 设置 return_exceptions=True 捕获异常对象而不中断
    results = await asyncio.gather(*tasks, return_exceptions=True)

    for i, res in enumerate(results, start=1):
        if isinstance(res, Exception):
            print(f"[{i}] 发生异常: {type(res).__name__} - {res}")
        else:
            print(f"[{i}] 正常结果: {res}")

if __name__ == "__main__":
    asyncio.run(main())
