import asyncio

async def fetch_data(task_id: int):
    await asyncio.sleep(0.1)
    if task_id == 2:
        raise ValueError(f"任务 {task_id} 数据解析失败")
    if task_id == 3:
        raise ConnectionError(f"任务 {task_id} 连接超时")
    return f"任务 {task_id} 的结果"

async def main():
    tasks = [fetch_data(i) for i in range(1, 5)]
    
    # 启用 return_exceptions=True，异常会作为普通对象返回
    results = await asyncio.gather(*tasks, return_exceptions=True)

    successes = []
    failures = []

    for item in results:
        if isinstance(item, Exception):
            failures.append(item)
        else:
            successes.append(item)

    print("--- 成功任务 ---")
    for res in successes:
        print(f"✓ {res}")

    print("\n--- 失败任务 ---")
    for err in failures:
        print(f"✗ 类型: {type(err).__name__}, 信息: {err}")

if __name__ == "__main__":
    asyncio.run(main())
