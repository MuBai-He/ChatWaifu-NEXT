def append(name, row):
    text = json.dumps(row, ensure_ascii=False, default=str)
    assert all(secret not in text for secret in SECRETS)
    with (STAGE/name).open('a') as target:
        target.write(text+'\n')

async def diagnostics():
    config = PublicWebConfig(reader_provider='crawl4ai', crawl4ai_builtin_fallback=True, crawl4ai_endpoint=os.environ['CHATWAIFU_PUBLIC_WEB__CRAWL4AI_ENDPOINT'], crawl4ai_api_token=SecretStr(TOKEN) if TOKEN else None)
    reader = PublicWebReader(provider_config=config)
    schema = yaml.safe_load((ROOT/'skills/builtin/web-read/chatwaifu.yaml').read_text())['definition']['capabilities'][0]['output_schema']
    for label, focus in [('default_excerpt', None), ('focused_excerpt', 'New lithium battery-powered power bank specifications')]:
        args = {'url':'https://www.icao.int/news/new-power-bank-restrictions-will-safeguard-international-aviation','dns_resolver':'cloudflare','fresh':True,'max_characters':6000}
        if focus: args['focus'] = focus
        before = time.monotonic()
        row = {'key':label,'arguments':args,'scope':'Independent real Reader diagnostic, not a Runtime Skill receipt','ok':False}
        try:
            row['result'] = await reader.read(args)
            validate(row['result'], schema)
            row['ok'] = True
        except SkillExecutionError as error:
            row['error_code'] = error.structured.code
            row['error_details'] = error.structured.details
        row['latency_ms'] = round((time.monotonic()-before)*1000)
        append('reader-diagnostics.jsonl', row)
        print(json.dumps({'diagnostic':label,'ok':row['ok'],'provider':row.get('result',{}).get('provider'),'fallback':row.get('result',{}).get('provider_fallback'),'focus_matched':row.get('result',{}).get('focus_matched'),'error_code':row.get('error_code'),'latency_ms':row['latency_ms']}),flush=True)

asyncio.run(diagnostics())
http_count = 0
original_stream = httpx2.AsyncClient.stream
def capture(self, method, url, **kwargs):
    global http_count
    if method.upper() == 'POST' and str(url).endswith('/chat/completions'):
        assert http_count < 20, 'Bounded HTTP attempts exhausted'
        http_count += 1
        body = kwargs['json']
        assert body['model'] == 'gemini-3.8-flash-high'
        append('actual-provider-payloads.jsonl', {'http_attempt':http_count,'captured_at_utc':datetime.now(UTC).isoformat(),'payload':body,'payload_sha256':hashlib.sha256(json.dumps(body,ensure_ascii=False,sort_keys=True).encode()).hexdigest()})
    return original_stream(self,method,url,**kwargs)
httpx2.AsyncClient.stream = capture
