import hashlib,json,os,subprocess,sys
from pathlib import Path
stage=Path('/tmp/cw2-q02-v26-contract-runtime-20261004');root=Path('/tmp/cw2-q02-contract-position-v26-eval')
assert not (stage/'http-resume-exit.json').exists(), 'No repeated resumption'
manifest=json.loads((stage/'snapshot.json').read_text())['sha256'];assert all(hashlib.sha256((root/n).read_bytes()).hexdigest()==h for n,h in manifest.items())
env=os.environ.copy();env['PYTHONPATH']=os.pathsep.join([str(root/'services/runtime/src'),str(root/'packages/protocol-python/src'),str(root/'packages/model-worker-sdk-python/src'),'/tmp/cw2-q02-eval-deps',str(root)])
jobs=json.loads((stage/'commands.json').read_text());codes={}
for i in range(2,6):
    job=jobs[i];output=Path(job['output'])
    failed=[json.loads(l) for l in (output/'incomplete.jsonl').read_text().splitlines()]
    assert len(failed)==1 and failed[0]['reason']=='HTTPStatusError'
    assert not (output/'results.jsonl').exists()
    with (stage/f'job-{i}-http-resume.log').open('w') as log:
        rc=subprocess.run([sys.executable,str(stage/'launch.py'),*job['arguments']],cwd=root,env=env,stdout=log,stderr=subprocess.STDOUT).returncode
    codes[str(i)]=rc
    print(json.dumps({'http_resume_job':i,'exit_code':rc}),flush=True)
    if rc not in {0,2}:break
(stage/'http-resume-exit.json').write_text(json.dumps({'codes':codes,'same_snapshot':True,'past_failures_retained':True,'budget_failed_instant_message_not_repeated':True},indent=2)+'\n')
