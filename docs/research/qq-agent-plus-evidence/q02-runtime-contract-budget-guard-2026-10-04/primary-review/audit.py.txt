from chatwaifu_runtime.providers.input_estimation import count_reference_tokens
root=Path('/tmp/cw2-q02-full-prepare/v26-contract-runtime/final-download')
def rows(path):return [json.loads(l) for l in path.read_text().splitlines()] if path.exists() else []
def percentile(values,q):
    if not values:return None
    ordered=sorted(values);p=(len(values)-1)*q;i=int(p);return round(ordered[i]+(ordered[min(i+1,len(values)-1)]-ordered[i])*(p-i),2)
records=[];totals=[];attempts=[];errors=[];omissions=[];missing=[];failures=[];incoming=[]
for output in sorted((root/'results').glob('*/*')):
    group=f'{output.parent.name}/{output.name}';replies=rows(output/'results.jsonl');totals.extend(replies)

        assert eligible
        call=max(eligible,key=lambda r:datetime.fromisoformat(r['started_at']));payload=w['payload'];messages=payload['messages'];checks={}

        checks['wire_hash']=hashlib.sha256(json.dumps(payload,ensure_ascii=False,sort_keys=True).encode()).hexdigest()==w['payload_sha256']
        checks['system_hash']=hashlib.sha256(messages[0]['content'].encode()).hexdigest()==call['system_prompt_sha256']
        pre=call['pre_user_system_prompt_sha256'];matches=[i for i,m in enumerate(messages) if m['role']=='system' and isinstance(m['content'],str) and hashlib.sha256(m['content'].encode()).hexdigest()==pre] if pre else []
        if pre:checks['pre_user_order']=len(matches)==1 and messages[matches[0]+1]['role']=='user'
        continuation=call['continuation_system_prompt_sha256']
        if continuation:checks['continuation_order']=messages[-1]['role']=='system' and hashlib.sha256(messages[-1]['content'].encode()).hexdigest()==continuation
        projection={k:payload[k] for k in ['messages','tools','tool_choice'] if k in payload}

        report=call['input_budget_report'];checks['complete_reference']=reference==report['estimated_input_tokens'];checks['input_allowance']=reference<=report['estimated_token_limit']
        expected=[]
        for e in call['input_tool_exchanges']:
            expected.append({'role':'assistant','content':e['assistant_text'] or None,'tool_calls':[{'id':c['call_id'],'type':'function','function':{'name':c['name'],'arguments':json.dumps(c['arguments'],ensure_ascii=False,separators=(',',':'))}} for c in e['calls']]})
            expected.extend({'role':'tool','tool_call_id':r['call_id'],'content':json.dumps(r['content'],ensure_ascii=False,separators=(',',':'))} for r in e['results'])
        actual=[m for m in messages if m['role']=='tool' or 'tool_calls' in m];checks['complete_tool_results']=actual==expected
        record={'group':group,'call_id':call['call_id'],'sample_key':call['sample_key'],'reference_tokens':reference,'checks':checks,'payload_sha256':w['payload_sha256']}

    for r in rows(output/'incomplete.jsonl'):
