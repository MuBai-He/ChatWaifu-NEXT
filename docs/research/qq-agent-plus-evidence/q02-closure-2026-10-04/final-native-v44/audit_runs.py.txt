def j(p): return json.loads(p.read_text())
def rows(p): return [json.loads(x) for x in p.read_text().splitlines() if x]
def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest()
def text_sha(s): return hashlib.sha256(s.encode()).hexdigest()
pending = {}
def persist(name, value):
    assert name not in pending
    pending[name] = value
def reference(payload):
    projection = {k: payload[k] for k in ("messages", "tools", "tool_choice", "response_format") if k in payload}
    return count_reference_tokens(json.dumps(projection, ensure_ascii=False, separators=(",", ":"))) + 16 * len(payload["messages"]) + 16 * len(payload.get("tools", []))
def percentile(values, p):
    a = sorted(values); index = (len(a) - 1) * p; lo = int(index); hi = min(lo + 1, len(a) - 1)
    return round(a[lo] + (a[hi] - a[lo]) * (index - lo), 3)
def originals(payload):
    result = []
    for message in payload["messages"]:
        content = message.get("content")
        if message["role"] == "tool":
            row = json.loads(content)
            data = row.get("data", {})
            if row.get("ok") and isinstance(data, dict) and isinstance(data.get("text"), str):
                result.append({"origin": "current_tool", "url": data["url"], "body_sha256": data["body_sha256"],
                               "text_sha256": text_sha(data["text"]), "characters": len(data["text"]), "truncated": data["truncated"]})
        elif isinstance(content, str) and content.startswith("[PUBLIC SOURCE DATA]\n"):
            for receipt in json.loads(content.split("\n", 1)[1])["receipts"]:
                data = receipt.get("data", {})
                if receipt["original_result"] == "available" and isinstance(data.get("text"), str):
                    result.append({"origin": "prior_receipt", "url": data["url"], "body_sha256": data["body_sha256"],
                                   "text_sha256": text_sha(data["text"]), "characters": len(data["text"]), "truncated": data["truncated"]})
    return result
def history_markers(payload):
    return [{"role": m["role"], "marker": m["content"]} for m in payload["messages"]
            if m["role"] in ("user", "assistant") and isinstance(m.get("content"), str)
            and m["content"].startswith("[Runtime omitted an earlier ")]

reports = {}
for label, directory, journal_name, group_turns, limit in (
    ("native-four-turn", native, "runtime-pilot/provider-rounds.jsonl", flows, 21370),
    ("frozen-guard-control", frozen, "provider-rounds.jsonl", turns, 22341),
):
    journal = rows(directory / journal_name)
    started = [r for r in journal if r["event"] == "started"]
    finished = [r for r in journal if r["event"] == "finished"]
    assert {r["call_id"] for r in started} == {r["call_id"] for r in finished}
    sent, http = rows(directory / "actual-provider-payloads.jsonl"), rows(directory / "http-outcomes.jsonl")
    assert len(started) == len(sent) == len(http)
    observations = []
    for wire, call, outcome in zip(sent, finished, http, strict=True):
        if "call_id" in wire: assert wire["call_id"] == call["call_id"]
        assert wire["sample_key"] == call["sample_key"]
        if "sample_key" in outcome: assert outcome["sample_key"] == wire["sample_key"]
        else: assert outcome["call_id"] == call["call_id"]
        payload = wire["payload"]
        assert payload["model"] == "gemini-3.8-flash-high" and payload["max_tokens"] == 8192
        assert text_sha(json.dumps(payload, ensure_ascii=False, sort_keys=True)) == wire["payload_sha256"]
        used, report = reference(payload), call["input_budget_report"]
        assert report["estimated_input_tokens"] == used <= report["estimated_token_limit"] == limit
        usage = call["usage"]
        observations.append({"sample_key": wire["sample_key"], "call_id": call["call_id"], "http_attempt": wire["http_attempt"],
            "reference_tokens": used, "input_limit": limit, "usage": usage,
            "reference_bias_percent": round((used - usage["prompt_tokens"]) / usage["prompt_tokens"] * 100, 6) if usage and usage.get("prompt_tokens") else None,
            "reported_omitted_history_indices": report["omitted_history_indices"], "actual_wire_history_markers": history_markers(payload),
            "reported_omitted_preamble_indices": report["omitted_tool_preamble_indices"], "read_originals": originals(payload),
            "http_elapsed_ms": outcome["elapsed_ms"]})
    known = [r["usage"] for r in observations if r["usage"] and all(type(r["usage"].get(k)) is int for k in ("prompt_tokens", "completion_tokens", "total_tokens"))]
    reasoning = [u["reasoning_tokens"] for u in known if type(u.get("reasoning_tokens")) is int]
    biases = [r["reference_bias_percent"] for r in observations if r["reference_bias_percent"] is not None]
    accounting = {"actual_http_attempts": len(sent), "logical_provider_requests": len(started), "known_usage_calls": len(known),
        "unknown_usage_calls": len(sent) - len(known), "known_usage_subtotals": {k: sum(u[k] for u in known) for k in ("prompt_tokens", "completion_tokens", "total_tokens")},
        "reasoning_known_calls": len(reasoning), "reasoning_unknown_calls": len(sent) - len(reasoning), "reasoning_known_subtotal": sum(reasoning),
        "reference_range": [min(r["reference_tokens"] for r in observations), max(r["reference_tokens"] for r in observations)],
        "bias_percent_range": [min(biases), max(biases)], "http_status_counts": dict(Counter(str(r["status"]) for r in http)),
        "http_error_counts": dict(Counter(r["error_type"] for r in http if r["error_type"])),
        "http_latency_p50_p95_ms": [percentile([r["elapsed_ms"] for r in http], p) for p in (.5, .95)],
        "turn_latency_p50_p95_ms": [percentile([r["elapsed_ms"] for r in group_turns], p) for p in (.5, .95)],
        "execution_counts": dict(Counter(r["execution"] for r in group_turns)), "first_visible_latency": None, "usd_cost": None,
        "native_context_maximum": None, "native_output_maximum": None}
    persist(label + "-wire-audit.json", observations)
    persist(label + "-accounting.json", accounting)
    reports[label] = accounting
