def save(name, value, append=False):
    text = json.dumps(value, ensure_ascii=False, default=str)
    assert all(secret not in text for secret in SECRETS)
    with (STAGE / name).open("a" if append else "w") as target:
        target.write(text + "\n")


def tree_manifest():
    # The admitted isolated Python source tree, not production or credential stores.
    return {str(p.relative_to(ROOT)): hashlib.sha256(p.read_bytes()).hexdigest()
            for p in sorted(ROOT.rglob("*.py"))
            if not any(part.startswith(".") or part == "__pycache__" for part in p.relative_to(ROOT).parts)}

def count_capture(request):
    global last_counted
    last_counted = request
    return original_estimate(request)

def guard_payload(request):
    return build_chat_completions_payload(MODEL, request, build_messages(request), request_usage=True)

def capture_guard(request, **kwargs):
    global last_counted
    last_counted = None
    row = {"sample_key": current["sample_key"], "actual_runtime_guard": True,
           "request": asdict(request), "payload": guard_payload(request), "dispatched": False}
    try:
        fitted = original_guard(request, **kwargs)
        row.update(outcome="fits", fitted_request=asdict(fitted),
                   fitted_payload=guard_payload(fitted), report=asdict(fitted.input_budget_report)
                   if fitted.input_budget_report else None)
        return fitted
    except input_budget.InputBudgetExceeded as error:
        assert last_counted is not None
        row.update(outcome="blocked", report=asdict(error.report),
                   fitted_request=asdict(last_counted), fitted_payload=guard_payload(last_counted))
        raise
    finally:
        save("guard-observations.jsonl", row, append=True)

async def capture_run(self, request, **kwargs):
    global current
    assert current is None
    current = {"sample_key": kwargs["sample_key"], "generation_id": str(request.generation_id)}
    row = {**current, "reply": None, "error_type": None, "runtime_fallback_events": []}
    started = time.monotonic()
    try:
        outcome = await original_run(self, request, **kwargs)
        row["reply"] = outcome.reply
        row["helper_reply_origin"] = outcome.reply_origin
        row["execution"] = "runtime_fallback" if row["runtime_fallback_events"] else "provider"
        return outcome
    except BaseException as error:
        row["execution"] = "failed"
        row["error_type"] = type(error).__name__
        raise
    finally:
        row["runtime_fallback_events"] = list(current.get("fallback_events", []))
        if row["execution"] != "failed":
            row["execution"] = "runtime_fallback" if row["runtime_fallback_events"] else "provider"
        row["trace"] = self.last_trace
        row["elapsed_ms"] = round((time.monotonic() - started) * 1000)
        save("flow-observations.jsonl", row, append=True)
        current = None


def capture_fallback(text, turn):
    assert current is not None
    names = [name for name, value in vars(tool_calling).items()
             if name.endswith("_REPLY") and isinstance(value, str) and value == text]
    assert names
    current.setdefault("fallback_events", []).append(names[0])
    return original_fallback(text, turn)


async def capture_recorded_stream(self, request):
    global recorded_provider, logical_count
    assert current is not None and recorded_provider is None
    logical_count += 1
    recorded_provider = self
    row = {"sample_key": current["sample_key"], "logical_request": logical_count,
           "reference_input_tokens": estimate_reference_input_tokens(request),
           "input_budget": asdict(request.input_budget) if request.input_budget else None,
           "input_budget_report": asdict(request.input_budget_report) if request.input_budget_report else None,
           "error_type": None}
    started = time.monotonic()
    try:
        async with contextlib.aclosing(original_recorded_stream(self, request)) as stream:
            async for event in stream:
                yield event
    except BaseException as error:
        row["error_type"] = type(error).__name__
        raise
    finally:
        row["call_id"] = self.calls[-1]["call_id"] if self.calls else None
        row["elapsed_ms"] = round((time.monotonic() - started) * 1000)
        save("wire-budget-observations.jsonl", row, append=True)
        recorded_provider = None


@contextlib.asynccontextmanager
async def capture_http(self, method, url, **kwargs):
    global http_count
    if str(url) != ENDPOINT + "/chat/completions":
        async with original_http_stream(self, method, url, **kwargs) as response:
            yield response
        return
    assert args.execute and current is not None and recorded_provider is not None
    if http_count >= HTTP_LIMIT:
        raise RuntimeError("Actual HTTP admission limit reached")
    body = kwargs["json"]
    assert body["model"] == MODEL and body.get("max_tokens") == 8192
    assert not body.get("response_format")
    assert body.get("stream_options", {}).get("include_usage") is True
    http_count += 1
    attempt = http_count
    call_id = recorded_provider.calls[-1]["call_id"]
    save("actual-provider-payloads.jsonl", {
        "sample_key": current["sample_key"], "http_attempt": attempt, "call_id": call_id,
        "payload": body, "payload_sha256": hashlib.sha256(
            json.dumps(body, ensure_ascii=False, sort_keys=True).encode()).hexdigest(),
    }, append=True)
    row = {"http_attempt": attempt, "call_id": call_id, "status": None, "error_type": None}
    started = time.monotonic()
    try:
        async with original_http_stream(self, method, url, **kwargs) as response:
            row["status"] = response.status_code
            yield response
    except BaseException as error:
        row["error_type"] = type(error).__name__
        raise
    finally:
        row["elapsed_ms"] = round((time.monotonic() - started) * 1000)
        save("http-outcomes.jsonl", row, append=True)


async def capture_send(self, request, **kwargs):
    # Only timing/status/category; no headers, credentials, or query strings.
    host, port = request.url.host, request.url.port
    kind = ("searxng" if host == "127.0.0.1" and port == 18080 else
            "crawl4ai" if host == "127.0.0.1" and port == 11235 else
            "cloudflare_dns" if host in {"cloudflare-dns.com", "1.1.1.1"} else None)
    if kind is None:
        return await original_send(self, request, **kwargs)
    row = {"sample_key": current["sample_key"] if current else None,
           "kind": kind, "status": None, "error_type": None}
    started = time.monotonic()
    try:
        response = await original_send(self, request, **kwargs)
        row["status"] = response.status_code
        return response
    except BaseException as error:
        row["error_type"] = type(error).__name__
        raise
    finally:
        row["elapsed_ms"] = round((time.monotonic() - started) * 1000)
        row["timing_scope"] = "send return; streamed bodies may not yet be consumed"
        save("source-http-observations.jsonl", row, append=True)
