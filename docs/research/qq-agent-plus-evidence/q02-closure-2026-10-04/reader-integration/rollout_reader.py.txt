def preserved_tree(text):
    tree = ast.parse(text)
    for node in tree.body:
        if isinstance(node, ast.ClassDef) and node.name == 'Crawl4aiReaderClient':
            node.body = [n for n in node.body if not
                         (isinstance(n, ast.AsyncFunctionDef) and n.name in {'read', '_read_pdf'})]
    return ast.dump(tree)

def command(args, limit=60):
    proc = subprocess.run(args, cwd=DEPLOY, capture_output=True, text=True, timeout=limit)
    output = proc.stdout + proc.stderr
    for value in secret_values:
        output = output.replace(value, '[REDACTED]')
    commands.append({'argv': args, 'exit_code': proc.returncode, 'output': output[-3000:]})
    if proc.returncode:
        raise RuntimeError('rollout_command_failed')


def atomic_copy(data, target):
    mode = stat.S_IMODE(target.stat().st_mode)
    temporary = target.with_name(target.name + '.q02-new')
    assert not temporary.exists()
    fd = os.open(temporary, os.O_WRONLY | os.O_CREAT | os.O_EXCL, mode)
    try:
        with os.fdopen(fd, 'wb') as file:
            file.write(data)
        os.replace(temporary, target)
    finally:
        if temporary.exists():
            temporary.unlink()


def health():
    context = ssl.create_default_context(cafile='/etc/ssl/certs/ca-certificates.crt')
    with socket.create_connection(('127.0.0.1', 8765), timeout=5) as raw:
        with context.wrap_socket(raw, server_hostname='mubai.website') as conn:
            conn.sendall(b'GET /v1/runtime/health HTTP/1.1\r\nHost: mubai.website:18443\r\nConnection: close\r\n\r\n')
            response = http.client.HTTPResponse(conn)
            response.begin()
            body = json.loads(response.read(16384))
            return {'http': response.status, 'status': body.get('status'),
                    'database': body.get('database'), 'certificate_verified': True}


def crawler_health():
    opener = urllib.request.build_opener(urllib.request.ProxyHandler({}))
    with opener.open('http://127.0.0.1:11235/health', timeout=5) as response:
        body = json.loads(response.read(4096))
        return {'http': response.status, 'status': body.get('status'), 'version': body.get('version')}
