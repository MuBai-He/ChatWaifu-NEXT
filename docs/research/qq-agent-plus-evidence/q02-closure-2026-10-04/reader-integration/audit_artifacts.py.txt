"""Read-only reader integration audit; no model, retrieval or deployment calls."""

import argparse
import ast
import hashlib
import json
from pathlib import Path


def load(path):
    return json.loads(path.read_text())


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def text_sha(value):
    return hashlib.sha256(value.encode()).hexdigest()


def unchanged_ast(path):
    tree = ast.parse(path.read_text())
    for node in tree.body:
        if isinstance(node, ast.ClassDef) and node.name == "Crawl4aiReaderClient":
            node.body = [item for item in node.body
                         if not isinstance(item, ast.AsyncFunctionDef)
                         or item.name not in {"read", "_read_pdf"}]
    return ast.dump(tree, include_attributes=False)


parser = argparse.ArgumentParser()
parser.add_argument("--write-report", action="store_true")
parser.add_argument("--check-current", action="store_true")
args = parser.parse_args()
phase = Path(__file__).resolve().parent
repo = phase.parents[4]
prior_counts = {}
for name, expected in load(phase / "prior-versions.json").items():
    previous = phase.parent / name
    assert sha(previous / "manifest.json") == expected, name
    files = load(previous / "manifest.json")["files"]
    for rel, digest in files.items():
        assert sha(previous / rel) == digest, (name, rel)
    prior_counts[name] = len(files)
assert prior_counts == {
    "revision-pipeline": 84, "scope-blocks": 93, "deployed-retrieval-check": 31}
documents = load(phase / "document-sha256.json")["files"]
assert len(documents) == 5
for rel, expected in documents.items():
    assert sha(phase / "final-documents" / (rel.replace("/", "__") + ".txt")) == expected
    if args.check_current:
        assert sha(repo / rel) == expected, rel

assert ast.dump(ast.parse(initial.read_text()), include_attributes=False) == ast.dump(
    ast.parse(after.read_text()), include_attributes=False)
