class CaptureStream(httpx2.AsyncByteStream):
 def __init__(self,original,owner):self.original=original;self.owner=owner
 async def __aiter__(self):
  async for block in self.original:self.owner.blocks.append(block);yield block
 async def aclose(self):await self.original.aclose()
class CaptureTransport(httpx2.AsyncBaseTransport):
 def __init__(self,identity):self.identity=identity;self.inner=httpx2.AsyncHTTPTransport();self.blocks=[];self.attempts=0;self.status=None
 async def handle_async_request(self,request):
  self.attempts+=1
  if self.attempts>1:raise RuntimeError('controlled_replay_disallows_second_HTTP_attempt')
  append('http-attempts.jsonl',{'key':self.identity,'payload':json.loads(request.content),'started_at':datetime.now(UTC).isoformat()})
  response=await self.inner.handle_async_request(request);self.status=response.status_code
  if response.status_code==200:response.stream=CaptureStream(response.stream,self)
  return response
 async def aclose(self):await self.inner.aclose()
