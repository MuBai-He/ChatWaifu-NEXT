def sha(data):
    return hashlib.sha256(data if isinstance(data, bytes) else data.encode()).hexdigest()
def write(name, data):
    (ROOT/name).write_text(json.dumps(data, ensure_ascii=False, indent=2, default=str)+'\n')
def append(name, data):
    with (ROOT/name).open('a') as f:
        f.write(json.dumps(data, ensure_ascii=False, default=str)+'\n')
def owner_route():
    owner=Path.home()/'Library/Application Support/local.chatwaifu.next/runtime'
    with sqlite3.connect(f'file:{owner}/chatwaifu.db?mode=ro', uri=True) as db:
        url=db.execute("SELECT base_url FROM model_role_configs WHERE role='chat'").fetchone()[0]
    assert sha(url)==json.loads((ROOT/'endpoint-metadata.json').read_text())['base_url_sha256']
    return url, LocalModelSecretStore(owner/'model-secrets.json').get('chat')
def check_freeze():
    plan=json.loads((ROOT/'plan.json').read_text())
    assert subprocess.check_output(['git','rev-parse','HEAD'], text=True).strip()==plan['code_commit']
    assert not subprocess.check_output(['git','status','--porcelain'], text=True).strip()
    assert sha(Path(__file__).read_bytes())==plan['driver_sha256']
    for item in plan['source_hashes']:
        assert sha((REPO/item['path']).read_bytes())==item['sha256']
    return plan

class Models:
    def __init__(self, window):
        self.window=window
        self.summary_inputs=[]
    def get(self, role):
        return SimpleNamespace(context_window=self.window)
    async def complete(self, role, system, user, **kwargs):
        # Deliberately controlled summary, rather than variable model summarization.
        # No missing fact is reconstructed. Its loss is separately reported.
        self.summary_inputs.append({'role':role, 'system':system, 'user':user})
        return '较早资料没有形成可引用的逐字记录；无法从摘要恢复原始编号。'

async def probe():
    check_freeze();url,key=owner_route()
    async with httpx2.AsyncClient(timeout=90) as client:
      for model in ['gemini-3.1-flash-lite','gemini-3.8-flash-high']:
        for maximum in [65536,65537]:
          payload={'model':model,'messages':[{'role':'user','content':'Reply exactly OK.'}],'stream':False,'max_tokens':maximum}
          start=time.perf_counter()
          r=await client.post(openai_compatible_endpoint(url,'chat/completions'),headers={'Authorization':f'Bearer {key}'},json=payload)
          data=r.json()
          # Do not write upstream error prose, which can contain private routes.
          item={'model':model,'requested_max_tokens':maximum,'status':r.status_code,'elapsed_ms':round((time.perf_counter()-start)*1000),'usage':data.get('usage'),'response_model':data.get('model'),'choices':data.get('choices') if r.status_code==200 else None,'error_type':data.get('error',{}).get('type')}
          append('output-capability-probes.jsonl',item); print(json.dumps(item),flush=True)

async def run():
    plan=check_freeze();url,key=owner_route()
    assert sha((ROOT/'requests.json').read_bytes())==plan['requests_sha256']
    requests=json.loads((ROOT/'requests.json').read_text())
    completed={json.loads(l)['key'] for l in (ROOT/'replies.jsonl').read_text().splitlines()} if (ROOT/'replies.jsonl').exists() else set()
    jobs=[(m,r,i) for m in plan['models'] for r in requests for i in range(plan['repeats'])]
    random.Random(20261001).shuffle(jobs)
    sem=asyncio.Semaphore(3)
    async def one(model,record,repetition):
      identity=f"{model}:{record['case']}:{record['variant']}:{repetition}"
      if identity in completed:return
      async with sem:
        if record['local_budget_error']:
            append('replies.jsonl',{'key':identity,'model':model,'case':record['case'],'variant':record['variant'],'repetition':repetition,'local_budget_error':True});return
        payload=record['payload']|{'model':model}
        start=time.perf_counter();first=None;chunks=[];usage=None;finish=None;error=None;status=None;response_model=None
        append('http-attempts.jsonl',{'key':identity,'payload':payload,'payload_sha256':sha(json.dumps(payload,ensure_ascii=False,separators=(',',':'))),'started_at':datetime.now(UTC).isoformat()})
        try:
          async with httpx2.AsyncClient(timeout=180) as client:
            async with client.stream('POST',openai_compatible_endpoint(url,'chat/completions'),headers={'Authorization':f'Bearer {key}'},json=payload) as resp:
              status=resp.status_code
              if status!=200: error='http_error';await resp.aread()
              else:
                async for line in resp.aiter_lines():
                  if not line.startswith('data:'):continue
                  value=line[5:].strip()
                  if value=='[DONE]':break
                  data=json.loads(value)
                  append('stream-events.jsonl',{'key':identity,'data':data})
                  response_model=data.get('model',response_model)
                  if data.get('usage'):usage=data['usage']
                  for choice in data.get('choices',[]):
                    text=choice.get('delta',{}).get('content')
                    if text:
                      if first is None:first=round((time.perf_counter()-start)*1000)
                      chunks.append(text)
                    if choice.get('finish_reason'):finish=choice['finish_reason']
        except Exception as e:error=type(e).__name__
        reply=''.join(chunks)
        normalized=reply.strip()
        if normalized.startswith('```'):normalized=normalized.split('\n',1)[-1].rsplit('```',1)[0].strip()
        try: parsed=json.loads(normalized)
        except Exception:parsed=None
        fact=parsed.get('fact') if isinstance(parsed,dict) else None
        item={'key':identity,'model':model,'response_model':response_model,'case':record['case'],'variant':record['variant'],'repetition':repetition,'status':status,'error':error,'reply':reply,'parsed':parsed,'fact_available':record['fact_available'],'fact_covered':fact==record['gold'],'unsupported_fact':not record['fact_available'] and fact is not None,'available_fact_error':record['fact_available'] and fact!=record['gold'],'reference_input_tokens':record['reference_tokens'],'usage':usage,'finish_reason':finish,'ttft_ms':first,'elapsed_ms':round((time.perf_counter()-start)*1000)}
        append('replies.jsonl',item);print(json.dumps({k:item[k] for k in ['key','status','error','fact_covered','fact_available','finish_reason','elapsed_ms']}),flush=True)
    await asyncio.gather(*(one(*j) for j in jobs))

if __name__=='__main__':
    asyncio.run({'prepare':prepare,'probe':probe,'run':run}[sys.argv[1]]())
