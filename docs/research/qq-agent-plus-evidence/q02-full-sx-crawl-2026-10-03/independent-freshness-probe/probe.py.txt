import asyncio,json,sys,time
from pathlib import Path
root=Path('/tmp/cw2-q02-source-revision-v7-eval')
sys.path[:0]=[str(root/'services/runtime/src'),str(root/'packages/protocol-python/src'),str(root/'packages/model-worker-sdk-python/src'),'/tmp/cw2-q02-eval-deps']
from pydantic import SecretStr
from chatwaifu_runtime.config.settings import PublicWebConfig
from chatwaifu_runtime.runtime_skills.public_web import PublicWebReader
from chatwaifu_runtime.runtime_skills.public_web_search import PublicWebSearch
from chatwaifu_runtime.runtime_skills.errors import SkillExecutionError
settings={}
for line in Path('/home/mubai/.local/share/chatwaifu-server/config/public-web.env').read_text().splitlines():
 if line.startswith('CHATWAIFU_PUBLIC_WEB__') and '=' in line:
  k,v=line.split('=',1);settings[k]=v.strip().strip(chr(34)).strip(chr(39))
config=PublicWebConfig(search_provider='searxng',searxng_endpoint=settings.get('CHATWAIFU_PUBLIC_WEB__SEARXNG_ENDPOINT','http://127.0.0.1:18080'),reader_provider='crawl4ai',crawl4ai_endpoint='http://127.0.0.1:11235',crawl4ai_api_token=SecretStr(settings['CHATWAIFU_PUBLIC_WEB__CRAWL4AI_API_TOKEN']),timeout_seconds=25)
reader=PublicWebReader(provider_config=config);search=PublicWebSearch(reader,provider_config=config)
out=Path('/tmp/cw2-q02-full-sx-crawl-20261003/independent-freshness-probe');out.mkdir(exist_ok=True)
async def main():
 rows=[]
 for label,args,fn in [('generic_current',{'query':'2026 国内航班 充电宝 最新 规定','max_results':5,'dns_resolver':'cloudflare'},search.search),('airline_current',{'query':'site:airchina.com.cn 2026 充电宝','max_results':5,'dns_resolver':'cloudflare'},search.search),('known_current_html',{'url':'https://www.airchina.com.cn/zh-CN/content/c/2026-08-04/30035.shtml','dns_resolver':'cloudflare','fresh':True,'max_characters':6000},reader.read)]:
  row={'label':label,'arguments':args};start=time.monotonic()
  try:row.update(ok=True,result=await fn(args))
  except SkillExecutionError as e:row.update(ok=False,error_code=e.structured.code,retryable=e.structured.retryable)
  row['latency_ms']=round(1000*(time.monotonic()-start));rows.append(row)
  (out/'results.json').write_text(json.dumps(rows,ensure_ascii=False,indent=2)+'\n')
  print(json.dumps({k:v for k,v in row.items() if k!='result'},ensure_ascii=False),flush=True)
asyncio.run(main())
